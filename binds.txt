# Пример файла binds.txt для GitHub. Формат такой же, как у binder.txt.
# Клавиши: имя (X, F5, NUMPAD1, VK_1, VK_MENU) или число (код VK). Одиночная цифра = код, поэтому клавишу "1" пишите VK_1.
# pinned=1  -> бинд закреплён вверху списка у друзей. Без pinned - идёт ниже закреплённых, но выше личных биндов друга.
# id=       -> уникальное имя бинда (по нему запоминаются клавиши, которые друг переназначил). Не меняйте его после выдачи.
# Сохраняйте файл в UTF-8 (по умолчанию на GitHub так и есть).

[bind]
id=domkrat
name=Домкрат
key=VK_1
key2=0
cmd=dk
enabled=1
pinned=1
line=800|/domkrat

[bind]
id=drugs
name=Наркотики
key=VK_3
key2=0
cmd=
enabled=1
pinned=0
line=800|/usedrugs 7
